	\section{Introduction to Vector Spaces}
	
	
	The concept of a \emph{vector space} is one of the most fundamental and unifying ideas
	in all of mathematics. While the geometric intuition of vectors as ``arrows in space''
	is useful, the formal algebraic structure captures a far richer family of objects
	--- including matrices, continuous functions, and polynomials --- under a single
	coherent framework.
	
	\begin{remark}{Motivation}{mot}
		Before introducing the abstract definition, consider the following objects:
		\begin{itemize}[leftmargin=2em]
			\item Ordered $n$-tuples of real numbers: $\R^n$.
			\item The set of all $m \times n$ real matrices: $\R^{m \times n}$.
			\item The set of all continuous functions on $[a,b]$: $C[a,b]$.
			\item The set of all polynomials of degree less than $n$: $P_n$.
		\end{itemize}
		Each of these admits natural operations of \emph{addition} and \emph{scalar
			multiplication} that obey the same algebraic laws.  A vector space is precisely
		the abstract object that encodes those shared laws.
	\end{remark}
	

	\section{Euclidean Vector Spaces}
	
	
	\subsection{The Space \texorpdfstring{$\R^n$}{Rn}}
	
	\begin{definition}{Euclidean $n$-Space}{euclidean}
		The \textbf{Euclidean $n$-space} $\R^n$ is the set of all ordered $n$-tuples of
		real numbers:
		\[
		\R^n \;=\; \bigl\{\, (x_1, x_2, \ldots, x_n) \;\big|\; x_i \in \R \text{ for all } i \bigr\}.
		\]
		Equivalently, every element of $\R^n$ may be identified with an $n \times 1$
		column vector
		\[
		\vv{x} = \begin{pmatrix} x_1 \\ x_2 \\ \vdots \\ x_n \end{pmatrix}.
		\]
		\textbf{Addition} and \textbf{scalar multiplication} are defined component-wise:
		\[
		\vv{x} + \vv{y}
		\;=\;
		\begin{pmatrix} x_1 + y_1 \\ x_2 + y_2 \\ \vdots \\ x_n + y_n \end{pmatrix},
		\qquad
		\alpha\vv{x}
		\;=\;
		\begin{pmatrix} \alpha x_1 \\ \alpha x_2 \\ \vdots \\ \alpha x_n \end{pmatrix},
		\quad \alpha \in \R.
		\]
	\end{definition}

	\subsection{The Space \texorpdfstring{$\R^{m\times n}$}{Rm x n}}

	
	\begin{definition}{Matrix Space $\R^{m\times n}$}{matspace}
		Let $\R^{m\times n}$ denote the set of all $m \times n$ matrices with real entries.
		For matrices $A = (a_{ij})$ and $B = (b_{ij})$ in $\R^{m\times n}$ and a scalar
		$\alpha \in \R$, define:
		\[
		(A + B)_{ij} \;=\; a_{ij} + b_{ij},
		\qquad
		(\alpha A)_{ij} \;=\; \alpha\, a_{ij}.
		\]
		The \textbf{zero matrix} $\vzero \in \R^{m\times n}$ is the matrix with all entries
		equal to zero, and the \textbf{additive inverse} of $A$ is $-A = (-a_{ij})$.
	\end{definition}
	
	\noindent
	To make the definitions concrete, consider two $2\times 3$ matrices:
	\[
	A = \begin{pmatrix} 1 & -2 & 3 \\ 0 & 4 & -1 \end{pmatrix},
	\quad
	B = \begin{pmatrix} 2 & 0 & -1 \\ 5 & -3 & 2 \end{pmatrix}.
	\]
	Then
	\[
	A + B
	= \begin{pmatrix} 3 & -2 & 2 \\ 5 & 1 & 1 \end{pmatrix},
	\qquad
	3A
	= \begin{pmatrix} 3 & -6 & 9 \\ 0 & 12 & -3 \end{pmatrix}.
	\]
	
	\begin{remark}{Connection to $\R^n$}{conn}
		The space $\R^n$ is a special case of $\R^{m\times n}$: identifying every
		$\vv{x}\in\R^n$ with its column-vector representation gives the isomorphism
		$\R^n \cong \R^{n\times 1}$.  The operations of addition and scalar
		multiplication on $\R^n$ are then the matrix operations restricted to
		$n\times 1$ matrices.
	\end{remark}
	
	
	\section{The Vector Space Axioms}

	\noindent
	Having observed that $\R^n$ and $\R^{m\times n}$ share the same algebraic rules,
	we now state those rules as \emph{axioms} that define an abstract vector space.
	
	\begin{axiombox}{Vector Space Axioms}{vs}
		Let $V$ be a non-empty set equipped with two operations:
		\begin{itemize}[leftmargin=2em]
			\item \textbf{Addition:} $+ : V\times V \to V$, written $(x,y)\mapsto x+y$.
			\item \textbf{Scalar Multiplication:} $\cdot : \R\times V \to V$,
			written $(\alpha, x)\mapsto \alpha x$.
		\end{itemize}
		Then $V$ (together with these operations) is called a \textbf{vector space over
			$\R$} if the following eight axioms hold for all $x,y,z\in V$ and all
		$\alpha,\beta\in\R$:
		
		\medskip
		\begin{tabular}{>{\bfseries}p{1.2cm} p{4.5cm} p{8cm}}
			\toprule
			Label & Name & Statement \\
			\midrule
			A1 & Commutativity &
			$x + y = y + x$ \\[4pt]
			A2 & Associativity (add.) &
			$(x + y) + z = x + (y + z)$ \\[4pt]
			A3 & Zero vector &
			$\exists\, \vzero \in V$ such that $x + \vzero = x$ \\[4pt]
			A4 & Additive inverse &
			$\forall\, x \in V,\; \exists\, {-x}\in V$ such that $x + (-x) = \vzero$ \\[4pt]
			A5 & Distributivity (scalar) &
			$\alpha(x + y) = \alpha x + \alpha y$ \\[4pt]
			A6 & Distributivity (vector) &
			$(\alpha + \beta)x = \alpha x + \beta x$ \\[4pt]
			A7 & Associativity (scalar) &
			$(\alpha\beta)x = \alpha(\beta x)$ \\[4pt]
			A8 & Identity scalar &
			$1\cdot x = x$ \\
			\bottomrule
		\end{tabular}
		
		\medskip\noindent
		The elements of $V$ are called \textbf{vectors}; the elements of $\R$ are called
		\textbf{scalars}.
	\end{axiombox}
	
	\begin{remark}{Why Eight Axioms?}{whyeight}
		The eight axioms partition into two groups:
		\begin{itemize}[leftmargin=2em]
			\item \textbf{Additive axioms (A1--A4):} These say that $(V,+)$ is an
			\emph{abelian group}.
			\item \textbf{Scalar axioms (A5--A8):} These control how scalars interact
			with vectors.
		\end{itemize}
		Together they capture the essential ``linearity'' that makes vector spaces
		so powerful.
	\end{remark}

	The diagram below illustrates the relationships among the axioms:

	\section{Examples of Vector Spaces}

	
	\subsection{The Function Space \texorpdfstring{$C[a,b]$}{C[a,b]}}
	
	\begin{definition}{$C[a,b]$}{cab}
		Let $C[a,b]$ denote the set of all real-valued functions that are defined and
		continuous on the closed interval $[a,b]$:
		\[
		C[a,b] \;=\; \bigl\{\, f : [a,b] \to \R \;\big|\; f \text{ is continuous} \bigr\}.
		\]
		For $f, g \in C[a,b]$ and $\alpha \in \R$, define
		\[
		(f + g)(x) \;=\; f(x) + g(x), \qquad (\alpha f)(x) \;=\; \alpha f(x),
		\quad \forall\, x \in [a,b].
		\]
	\end{definition}
	
	\noindent
	\textbf{Verification of the axioms.}
	
	\begin{itemize}[leftmargin=2em, itemsep=0.4em]
		\item \textbf{Closure under addition:} The sum of two continuous functions is
		continuous, so $f+g \in C[a,b]$.
		\item \textbf{Closure under scalar multiplication:} $\alpha f$ is continuous
		since it is a scalar multiple of a continuous function.
		\item \textbf{A1 (Commutativity):} For every $x\in[a,b]$,
		\[
		(f+g)(x) = f(x)+g(x) = g(x)+f(x) = (g+f)(x),
		\]
		so $f+g = g+f$.
		\item \textbf{A3 (Zero vector):} The zero function $z(x)=0$ for all
		$x\in[a,b]$ satisfies $(f+z)(x) = f(x)+0 = f(x)$, hence $f+z=f$.
		\item \textbf{A4 (Additive inverse):} For any $f\in C[a,b]$, the function
		$(-f)(x) = -f(x)$ is continuous and satisfies
		$(f + (-f))(x) = f(x) - f(x) = 0 = z(x)$.
		\item Axioms A2, A5--A8 follow directly from the corresponding properties
		of real numbers applied pointwise.
	\end{itemize}

	\subsection{The Polynomial Space \texorpdfstring{$P_n$}{Pn}}

	\begin{definition}{Polynomial Space $P_n$}{polspace}
		Let $P_n$ denote the set of all polynomials of degree \emph{strictly less than}
		$n$:
		\[
		P_n \;=\; \bigl\{\, p(x) = a_{n-1}x^{n-1} + a_{n-2}x^{n-2} + \cdots
		+ a_1 x + a_0 \;\big|\; a_i \in \R \bigr\}.
		\]
		For $p, q \in P_n$ and $\alpha \in \R$, define
		\[
		(p+q)(x) \;=\; p(x) + q(x), \qquad (\alpha p)(x) \;=\; \alpha\, p(x).
		\]
		The \textbf{zero vector} is the zero polynomial
		\[
		z(x) = 0x^{n-1} + 0x^{n-2} + \cdots + 0x + 0.
		\]
	\end{definition}
	
	\begin{example}{Addition in $P_3$}{padd}
		Let $p(x) = 2x^2 - 3x + 1$ and $q(x) = -x^2 + 5x - 2$, both in $P_3$.
		Then:
		\begin{align*}
			(p+q)(x) &= (2x^2 - 3x + 1) + (-x^2 + 5x - 2) = x^2 + 2x - 1, \\
			(3p)(x)  &= 3(2x^2 - 3x + 1) = 6x^2 - 9x + 3.
		\end{align*}
		Both results lie in $P_3$, confirming closure.
	\end{example}

	\section{Verification of Axioms: Detailed Walkthrough}

	
	\noindent
	To illustrate how the axioms are verified, we walk through the full verification
	for $\R^{m\times n}$.
	
	\begin{example}{Verifying the axioms for $\R^{m\times n}$}{axverify}
		Let $A=(a_{ij}), B=(b_{ij}), C=(c_{ij}) \in \R^{m\times n}$ and
		$\alpha,\beta\in\R$.
		
		\begin{description}[leftmargin=2em, font=\bfseries\color{axiomblue}]
			\item[A1.] $(A+B)_{ij} = a_{ij}+b_{ij} = b_{ij}+a_{ij} = (B+A)_{ij}$.
			\item[A2.] $((A+B)+C)_{ij} = (a_{ij}+b_{ij})+c_{ij}
			= a_{ij}+(b_{ij}+c_{ij}) = (A+(B+C))_{ij}$.
			\item[A3.] The zero matrix $\vzero$ with all entries $0_{ij}=0$ satisfies
			$(A+\vzero)_{ij} = a_{ij}+0 = a_{ij}$.
			\item[A4.] $(-A)_{ij} = -a_{ij}$ gives $(A+(-A))_{ij} = a_{ij}-a_{ij} = 0$.
			\item[A5.] $(\alpha(A+B))_{ij} = \alpha(a_{ij}+b_{ij})
			= \alpha a_{ij}+\alpha b_{ij} = (\alpha A+\alpha B)_{ij}$.
			\item[A6.] $((\alpha+\beta)A)_{ij} = (\alpha+\beta)a_{ij}
			= \alpha a_{ij}+\beta a_{ij} = (\alpha A+\beta A)_{ij}$.
			\item[A7.] $((\alpha\beta)A)_{ij} = (\alpha\beta)a_{ij}
			= \alpha(\beta a_{ij}) = (\alpha(\beta A))_{ij}$.
			\item[A8.] $(1\cdot A)_{ij} = 1\cdot a_{ij} = a_{ij}$.
		\end{description}
		All eight axioms hold, confirming that $\R^{m\times n}$ is a vector space.
	\end{example}
	
	
	\section{Fundamental Properties of Vector Spaces}

	
	\noindent
	The eight axioms imply several additional useful properties.  The theorem below
	collects the three most basic ones and provides complete proofs.
	
	\begin{theorem}{Additional Properties of Vector Spaces (Theorem 3.1.1)}{thm311}
		Let $V$ be a vector space and let $x \in V$.  Then:
		\begin{enumerate}[label=(\roman*), leftmargin=2.5em]
			\item $0\, x = \vzero$.
			\item $x + y = \vzero \;\Longrightarrow\; y = -x$ \quad
			(the additive inverse is \emph{unique}).
			\item $(-1)\, x = -x$.
		\end{enumerate}
	\end{theorem}
	
	\begin{proof}
		\textbf{(i)} \quad By axioms \textbf{A6} and \textbf{A8}:
		\[
		x \;=\; 1\cdot x \;=\; (1+0)\cdot x \;=\; 1\cdot x + 0\cdot x \;=\; x + 0\cdot x.
		\]
		Adding $-x$ to both sides:
		\[
		\vzero \;=\; -x + x \;=\; -x + (x + 0\cdot x)
		\;\overset{\text{A2}}{=}\; (-x+x) + 0\cdot x
		\;\overset{\text{A4}}{=}\; \vzero + 0\cdot x
		\;\overset{\text{A3}}{=}\; 0\cdot x.
		\]
		
		\medskip
		\textbf{(ii)} \quad Suppose $x + y = \vzero$.  Then:
		\begin{align*}
			-x &\;=\; -x + \vzero
			\;\overset{\text{A3}}{=}\; -x + (x + y) \\
			&\;\overset{\text{A2}}{=}\; (-x+x) + y
			\;\overset{\text{A4}}{=}\; \vzero + y
			\;\overset{\text{A1, A3}}{=}\; y.
		\end{align*}
		
		\medskip
		\textbf{(iii)} \quad By part (i) and axiom \textbf{A6}:
		\[
		\vzero \;=\; 0\cdot x \;=\; (1+(-1))\cdot x
		\;\overset{\text{A6}}{=}\; 1\cdot x + (-1)\cdot x
		\;\overset{\text{A8}}{=}\; x + (-1)\cdot x.
		\]
		Thus $x + (-1)\cdot x = \vzero$, and by part (ii), $(-1)\cdot x = -x$.
	\end{proof}

	\section{Further Consequences of the Axioms}

	
	\noindent
	Beyond Theorem~\ref{thm:thm311}, the axioms yield the following important results.
	Their proofs are left as guided exercises.
	
	\begin{theorem}{Additional Consequences}{addcons}
		Let $V$ be a vector space, $\alpha \in \R$, and $x \in V$.
		\begin{enumerate}[label=(\roman*), leftmargin=2.5em]
			\item $\alpha\, \vzero = \vzero$.
			\item $\alpha\, x = \vzero \;\Longrightarrow\; \alpha = 0$ or $x = \vzero$.
			\item The zero vector $\vzero$ is unique.
		\end{enumerate}
	\end{theorem}
	
	\begin{proof}[Proof of (i)]
		By axiom A3, $\vzero + \vzero = \vzero$.  Multiplying by $\alpha$ and applying
		A5:
		\[
		\alpha\vzero = \alpha(\vzero + \vzero) = \alpha\vzero + \alpha\vzero.
		\]
		Adding $-(\alpha\vzero)$ to both sides (using A4 and A3):
		\[
		\vzero = \alpha\vzero.
		\]
	\end{proof}
	
	\begin{proof}[Proof of (iii) (Uniqueness of the zero vector)]
		Suppose $\vzero_1$ and $\vzero_2$ both satisfy A3.  Then:
		\[
		\vzero_1 \;=\; \vzero_1 + \vzero_2 \;=\; \vzero_2 + \vzero_1 \;=\; \vzero_2,
		\]
		where the first and third equalities use A3 for $\vzero_2$ and $\vzero_1$
		respectively, and the middle equality uses A1.
	\end{proof}
	
	
	\section{Summary}

	
	The key ideas of this chapter may be summarised as follows:
	
	\begin{tcolorbox}[
		enhanced, colback=shadeyellow, colframe=orange!70!black,
		title={\bfseries Chapter Summary}, fonttitle=\bfseries, breakable
		]
		\begin{itemize}[leftmargin=1.5em, itemsep=0.5em]
			\item A \textbf{vector space} over $\R$ is a set $V$ with addition and scalar
			multiplication satisfying axioms A1--A8.
			\item The Euclidean space $\R^n$, the matrix space $\R^{m\times n}$, the
			function space $C[a,b]$, and the polynomial space $P_n$ are all
			\textbf{concrete examples} of vector spaces.
			\item The \textbf{zero vector} is unique; every element has a \textbf{unique
				additive inverse}; and $(-1)x = -x$.
			\item The scalar $0$ always maps any vector to $\vzero$:
			$0\cdot x = \vzero$.
			\item The abstract framework unifies seemingly different mathematical objects
			under a single set of algebraic rules, enabling results proved once to
			apply universally.
		\end{itemize}
	\end{tcolorbox}
